fun calculadora(num1: Double, num2: Double, operacion: Char) {
    var resultado = 0.0

    if (operacion == '/' && num2 == 0.0) {
        println("Error: No se puede dividir por cero.")
        return
    }

    if (operacion == '+') {
        resultado = num1 + num2
    } else if (operacion == '-') {
        resultado = num1 - num2
    } else if (operacion == '*') {
        resultado = num1 * num2
    } else if (operacion == '/') {
        resultado = num1 / num2
    } else {
        println("Error: Operación no válida.")
        return
    }

    println("Resultado: $num1 $operacion $num2 = $resultado")
}

fun main() {
    calculadora(10.0, 5.0, '+')
    calculadora(10.0, 0.0, '/')
}
